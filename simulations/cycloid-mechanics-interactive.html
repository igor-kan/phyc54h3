Overview

The cycloid is one of the most celebrated curves in classical mechanics and the calculus of variations. Discovered by Galileo and investigated by Bernoulli, Newton, Leibniz, and Huygens, it possesses two remarkable variational properties:

  1. The Brachistochrone: The path of fastest descent under uniform gravity between two points.
  2. The Tautochrone (Isochrone): A curve along which a particle released from rest at any point reaches the bottom in the exact same time, independent of release height.
  3. The Cycloidal Pendulum: By suspending a flexible pendulum string between two cycloidal guides (“cheeks”), the string unwraps along the cheeks so the bob traces an exact evolute cycloid, eliminating the “circular error” of ordinary pendulums.

Simulation 3: The Tautochrone (Isochrone) Property

Huygens discovered in 1659 that an inverted cycloid is tautochronous: a frictionless particle sliding down a cycloidal track to the lowest point takes the exact same time regardless of where on the arch it is released!

The equation of motion along the arc length \(s\) (measured from the trough) is pure simple harmonic motion: \[ \ddot{s} + \left(\frac{g}{4a}\right) s = 0 \] All beads reach the trough \(s = 0\) at the identical quarter-period mark: \[ t_{\text{trough}} = \pi \sqrt{\frac{a}{g}} \]

Simulation 4: Huygens’ Isochronous Cycloidal Pendulum

In a standard circular pendulum, the period increases with amplitude: \[ T \approx 2\pi\sqrt{\frac{L}{g}}\left(1 + \frac{1}{16}\theta_0^2 + \frac{11}{3072}\theta_0^4 + \dots\right) \] This circular error caused grandfather clocks to lose time whenever the drive spring wound down and the swing amplitude decreased.

Christiaan Huygens eliminated this error in 1673 by flanking the suspension string with two curved metal guides shaped like cycloids (cycloidal cheeks). As the pendulum swings wide, the flexible string wraps against the cheek, shortening the effective radius of curvature and accelerating the bob just enough so that the trajectory of the bob is an exact evolute cycloid!

The period of Huygens’ cycloidal clock is strictly invariant across all amplitudes: \[ T_{\text{cycloid}} = 4\pi\sqrt{\frac{a}{g}} \]

(function() {
  const canvas = document.getElementById('wheel-canvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');

  let a = 40;
  let theta = 0;
  const maxTheta = 4 * Math.PI;
  let running = true;
  let animId = null;
  let lastTimestamp = 0;
  const rollSpeed = 1.6;

  const xOrigin = 50;
  const yRail = 45;

  function updateReadouts() {
    const rVal = document.getElementById('wheel-radius-val');
    const thVal = document.getElementById('wheel-theta-val');
    const thSlider = document.getElementById('wheel-theta-slider');
    if (rVal) rVal.innerText = a + ' px';
    if (thVal) thVal.innerText = theta.toFixed(2) + ' rad';

    const cReadout = document.getElementById('wheel-center-readout');
    const mReadout = document.getElementById('wheel-marker-readout');
    const aReadout = document.getElementById('wheel-angle-readout');
    const sReadout = document.getElementById('wheel-arc-readout');

    if (cReadout) cReadout.innerText = `(${Math.round(a*theta)}, ${a}) px`;
    if (mReadout) mReadout.innerText = `(${Math.round(a*(theta - Math.sin(theta)))}, ${Math.round(a*(1 - Math.cos(theta)))}) px`;

    const thMod = theta % (2 * Math.PI);
    const psiDeg = ((Math.PI - thMod) / 2) * (180 / Math.PI);
    let desc = "";
    if (Math.abs(psiDeg - 90) < 2) desc = "(Vertical drop)";
    else if (Math.abs(psiDeg) < 2) desc = "(Horizontal trough)";
    else if (Math.abs(psiDeg + 90) < 2) desc = "(Vertical rise)";
    if (aReadout) aReadout.innerText = `${psiDeg.toFixed(1)}° ${desc}`;

    const sArc = 4 * a * (1 - Math.cos(thMod / 2));
    if (sReadout) sReadout.innerText = `${sArc.toFixed(1)} px`;
  }

      ctx.beginPath();

  function step(ts) {

    rSlider.oninput = function() {
      a = parseInt(this.value);
      updateReadouts();
      draw();
    };
  }

  const thSlider = document.getElementById('wheel-theta-slider');
  if (thSlider) {
    thSlider.oninput = function() {
      theta = (parseInt(this.value) / 1256) * maxTheta;
      updateReadouts();
      draw();
    };
  }

  updateReadouts();
  draw();
  animId = requestAnimationFrame(step);
})();

(function() {
  const canvas = document.getElementById('brach-canvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');

  const g = 9.81;
  const xEnd = 620;
  const yEnd = 310;
  const xStart = 80;
  const yStart = 60;
  const W = xEnd - xStart;
  const H = yEnd - yStart;

  // Cycloid parameter solver for endpoint (W, H)
  // x = a*(theta - sin(theta)), y = a*(1 - cos(theta))
  // y/x = (1 - cos(theta)) / (theta - sin(theta))
  let thetaEnd = 3.5;
  for (let i = 0; i < 20; i++) {
    const f = (1 - Math.cos(thetaEnd)) / (thetaEnd - Math.sin(thetaEnd)) - (H / W);
    const num = Math.sin(thetaEnd) * (thetaEnd - Math.sin(thetaEnd)) - (1 - Math.cos(thetaEnd)) * (1 - Math.cos(thetaEnd));
    const den = Math.pow(thetaEnd - Math.sin(thetaEnd), 2);
    const fprime = num / den;
    thetaEnd -= f / fprime;
  }
  const aRadius = H / (1 - Math.cos(thetaEnd));

  // Build discretized paths
  const N = 400;
  function makePath(func) {
    const pts = [];
    let totLen = 0;
    for (let i = 0; i <= N; i++) {
      const u = i / N;
      const pt = func(u);
      if (i > 0) {
        const dx = pt.x - pts[i-1].x;
        const dy = pt.y - pts[i-1].y;
        totLen += Math.sqrt(dx*dx + dy*dy);
      }
      pts.push({ x: pt.x, y: pt.y, dist: totLen });
    }
    return { pts, totLen };
  }

    const th = u * thetaEnd;

  });

  const dToCenter = Math.sqrt(Math.max(0, R_circ*R_circ - (chord*chord)/4));
  const nx = -(yEnd - yStart) / chord;
  const ny = (xEnd - xStart) / chord;
  const circCenter = { x: mx - nx * dToCenter, y: my - ny * dToCenter };
  const ang1 = Math.atan2(yStart - circCenter.y, xStart - circCenter.x);
  const ang2 = Math.atan2(yEnd - circCenter.y, xEnd - circCenter.x);

  const circlePath = makePath(u => {
    const ang = ang1 + u * (ang2 - ang1);
    return { x: circCenter.x + R_circ * Math.cos(ang), y: circCenter.y + R_circ * Math.sin(ang) };

    const pts = path.pts;

    if (dist >= path.totLen) return pts[pts.length - 1];

      if (pts[mid].dist < dist) low = mid + 1;
      else high = mid - 1;

    return { x: p0.x + t * (p1.x - p0.x), y: p0.y + t * (p1.y - p0.y) };
  }

# Simulation 3: The Tautochrone (Isochrone) Property

Huygens discovered in 1659 that an inverted cycloid is **tautochronous**: a frictionless particle sliding down a cycloidal track to the lowest point takes the *exact same time* regardless of where on the arch it is released!

The equation of motion along the arc length $s$ (measured from the trough) is pure simple harmonic motion:
$$ \ddot{s} + \left(\frac{g}{4a}\right) s = 0 $$
All beads reach the trough $s = 0$ at the identical quarter-period mark:
$$ t_{\text{trough}} = \pi \sqrt{\frac{a}{g}} $$

In the simulation below, four beads are placed at widely disparate initial release heights along the cycloid. Press **"Release All Beads"** and watch them converge at the bottom in perfect synchronization!

# Simulation 4: Huygens' Isochronous Cycloidal Pendulum

In a standard circular pendulum, the period increases with amplitude:
$$ T \approx 2\pi\sqrt{\frac{L}{g}}\left(1 + \frac{1}{16}\theta_0^2 + \frac{11}{3072}\theta_0^4 + \dots\right) $$
This **circular error** caused grandfather clocks to lose time whenever the drive spring wound down and the swing amplitude decreased.

Christiaan Huygens eliminated this error in 1673 by flanking the suspension string with two curved metal guides shaped like cycloids (**cycloidal cheeks**). As the pendulum swings wide, the flexible string wraps against the cheek, shortening the effective radius of curvature and accelerating the bob just enough so that the trajectory of the bob is an exact evolute cycloid!

The period of Huygens' cycloidal clock is strictly invariant across all amplitudes:
$$ T_{\text{cycloid}} = 4\pi\sqrt{\frac{a}{g}} $$

    ctx.shadowBlur = 0;

    // -------------------------------------------------------------
    // 2. RIGHT: Simple Circular Pendulum
    // -------------------------------------------------------------
    // Ceiling
    ctx.fillStyle = '#475569';
    ctx.fillRect(pivot2.x - 70, pivot2.y - 12, 140, 12);
    ctx.fillStyle = '#cbd5e1';
    ctx.fillText("Standard Pivot", pivot2.x - 45, pivot2.y - 18);

    const bob2X = pivot2.x + L * 1.35 * Math.sin(theta_simple);
    const bob2Y = pivot2.y + L * 1.35 * Math.cos(theta_simple);

    // Rod
    ctx.lineWidth = 2;
    ctx.strokeStyle = '#e2e8f0';
    ctx.beginPath();
    ctx.moveTo(pivot2.x, pivot2.y);
    ctx.lineTo(bob2X, bob2Y);
    ctx.stroke();

    // Circular Bob
    ctx.shadowColor = '#f43f5e';
    ctx.shadowBlur = 12;
    ctx.fillStyle = '#f43f5e';
    ctx.beginPath();
    ctx.arc(bob2X, bob2Y, 12, 0, 2*Math.PI);
    ctx.fill();
    ctx.shadowBlur = 0;

    // Center pivot dots
    ctx.fillStyle = '#ffffff';
    ctx.beginPath(); ctx.arc(pivot1.x, pivot1.y, 4, 0, 2*Math.PI); ctx.fill();
    ctx.beginPath(); ctx.arc(pivot2.x, pivot2.y, 4, 0, 2*Math.PI); ctx.fill();
  }

  function step(ts) {
    if (!lastTs) lastTs = ts;
    const dt = Math.min((ts - lastTs) / 1000, 0.04);
    lastTs = ts;

    simTime += dt;

    // RK4 integration for non-linear simple pendulum
    const L_phys = (L * 1.35) / 120; // scale
    function f_alpha(th) { return -(g / L_phys) * Math.sin(th); }
    
    // Sub-stepping for stability
    const subSteps = 4;
    const subDt = dt / subSteps;
    for (let stepIdx = 0; stepIdx < subSteps; stepIdx++) {
      const k1_v = f_alpha(theta_simple);
      const k1_x = omega_simple;

      const k2_v = f_alpha(theta_simple + 0.5 * subDt * k1_x);
      const k2_x = omega_simple + 0.5 * subDt * k1_v;

      const k3_v = f_alpha(theta_simple + 0.5 * subDt * k2_x);
      const k3_x = omega_simple + 0.5 * subDt * k2_v;

      const k4_v = f_alpha(theta_simple + subDt * k3_x);
      const k4_x = omega_simple + subDt * k3_v;

      omega_simple += (subDt / 6) * (k1_v + 2*k2_v + 2*k3_v + k4_v);
      theta_simple += (subDt / 6) * (k1_x + 2*k2_x + 2*k3_x + k4_x);
    }

    render();

    if (running) {
      animId = requestAnimationFrame(step);
    }
  }

  function reset() {
    simTime = 0;
    lastTs = 0;
    theta_simple = ampRad;
    omega_simple = 0.0;
    render();
  }

  document.getElementById('huygens-toggle-btn').onclick = function() {
    running = !running;
    this.innerText = running ? '❚❚ Pause' : '▶ Resume';
    if (running) {
      lastTs = 0;
      animId = requestAnimationFrame(step);
    }
  };

  document.getElementById('huygens-reset-btn').onclick = function() {
    reset();
  };

  const slider = document.getElementById('huygens-amp-slider');
  slider.oninput = function() {
    ampDeg = parseInt(this.value);
    ampRad = ampDeg * Math.PI / 180;
    document.getElementById('huygens-amp-val').innerText = ampDeg + '°';
    updateSimplePendulumReadout();
    reset();
  };

  render();
  animId = requestAnimationFrame(step);
})();
</script>
